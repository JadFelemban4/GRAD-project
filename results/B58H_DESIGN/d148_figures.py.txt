def path(rel):

def text(rel):

def jload(rel):

def _value(node):
    """A literal, or the first argument of a call such as np.array([...], dtype=...)."""
    if isinstance(node, ast.Call) and node.args:
        return ast.literal_eval(node.args[0])
    return ast.literal_eval(node)


def consts(rel, names, count_only=(), source=None):
    """{name: (value, line)} of module-level assignments in rel, read by ast and never imported. A name in
    count_only gives the number of elements of its tuple. Refused if a name is missing or assigned twice."""
    tree = ast.parse(source if source is not None else text(rel))
    found = {}
    for node in tree.body:
        if not isinstance(node, ast.Assign):
            continue
        for t in node.targets:
            pairs = []
            if isinstance(t, ast.Name):
                pairs = [(t, node.value)]
            elif isinstance(t, ast.Tuple) and isinstance(node.value, ast.Tuple) and len(t.elts) == len(node.value.elts):
                pairs = list(zip(t.elts, node.value.elts))
            for el, v in pairs:
                if isinstance(el, ast.Name) and el.id in names:
                    if el.id in found:
                        raise Refused(f"{rel}: {el.id} is assigned twice at module level")
                    found[el.id] = (len(v.elts) if el.id in count_only else _value(v), node.lineno)
    missing = [n for n in names if n not in found]
    if missing:
        raise Refused(f"{rel}: no module-level {missing}")
    return found


def compiled(rel, name, env):
    """A module-level function or class of rel, compiled from its own source into a namespace holding env."""
    tree = ast.parse(text(rel))
    for node in tree.body:
        if isinstance(node, (ast.FunctionDef, ast.ClassDef)) and node.name == name:
            ns = dict(env)
            exec(compile(ast.Module(body=[node], type_ignores=[]), rel, "exec"), ns)
            return ns[name], node.lineno
    raise Refused(f"{rel}: no module-level {name}")


def budget_lines():
    """{label: (lo, hi, line)} of budget.txt's route table."""
    out = {}
    for i, line in enumerate(text("results/B58H_DESIGN/budget.txt").splitlines(), 1):
        m = re.match(r"^  (\S.*?)\s+(-?\d+\.\d\d)\s+(-?\d+\.\d\d)$", line)
        if m:
            out[m.group(1)] = (Fraction(m.group(2)), Fraction(m.group(3)), i)
    return out


def h(x):
    return f"{float(x):.2f} h"


def hr(lo, hi):
    return f"{float(lo):.2f}-{float(hi):.2f} h"


def tm(t):
    return t.strftime("%d %b %H:%M")


def hm(seconds):
    m = int(math.floor(abs(seconds) / 60))
    return f"{'-' if seconds < 0 else '+'}{m // 60}:{m % 60:02d}"

    p(f"  the training's: snapshot_steps({PROD}, {WINDOW}) = {snapshot_steps(PROD, WINDOW)}")

        f = LastThird(1.0)

    sighted = n * (ARMS - 1)

    practice = (HAND + PRACTICE_SEEDS * PRACTICE_ARMS) * N_PRACTICE

def checked(lines):
    """The output as text, refused unless every line is ASCII, holds no run of 12 or more digits, no 40- or
    64-character hex value and no guarded field's name."""
    for i, line in enumerate(lines, 1):
        if not line.isascii():
            raise Refused(f"output line {i} is not ASCII")
        if DIGITS.search(line):
            raise Refused(f"output line {i} holds a run of 12 or more digits")
        if HEX.search(line):
            raise Refused(f"output line {i} holds a 40- or 64-character hex value")
        if any(w in line for w in GUARDED):
            raise Refused(f"output line {i} names a guarded field of roads.json")
    return "".join(line + "\n" for line in lines)


def main():
    try:
        out = checked(build())
    except (Refused, OSError, KeyError, IndexError, TypeError, ValueError, AttributeError, ZeroDivisionError,
            SyntaxError) as e:
        sys.stderr.write(f"REFUSED: d148_figures.py: {type(e).__name__}: {e}\n")
        return 2
    sys.stdout.buffer.write(out.encode("ascii"))
    sys.stdout.buffer.flush()
    return 0


if __name__ == "__main__":
    sys.exit(main())
