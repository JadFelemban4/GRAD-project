def summ(rs, reading="noknock"):
    Dk = "D_noknock" if reading == "noknock" else "D_knock"
    Bk = "base_noknock" if reading == "noknock" else "base_knock"
    pts = [100.0 * (c[Dk] - r[Dk]) / r[Bk] for c, r in zip(cg, rs)]
    ev = 100.0 * (statistics.median(c[Dk] for c in cg) - statistics.median(r[Dk] for r in rs)) / statistics.median(x[Dk] for x in nt)
    fx = [100.0 * (r["fuel"] / c["fuel"] - 1.0) for c, r in zip(cg, rs)]
    return dict(mean=float(np.mean(pts)), ev=float(ev), fuel=float(np.mean(fx)), pts=pts, fx=fx,
                ret=float(np.mean([r["ret"] for r in rs])), rets=[r["ret"] for r in rs])

# ---------------------------------------------------------------- frontiers
def stair(pts):
    pts = sorted(pts)
    front, best = [(0.0, 0.0)], 0.0
    for x, y in pts:
        if y > best:
            front.append((x, y)); best = y
    return front


def hull(pts):
    """upper concave hull through (0,0), monotone part."""
    P = sorted(set([(0.0, 0.0)] + list(pts)))
    H = []
    for p in P:
        while len(H) >= 2:
            (x1, y1), (x2, y2) = H[-2], H[-1]
            # remove H[-1] if it lies on or below the chord H[-2] -> p
            if (y2 - y1) * (p[0] - x1) <= (p[1] - y1) * (x2 - x1):
                H.pop()
            else:
                break
        H.append(p)
    # keep the increasing part
    out = [H[0]]
    for p in H[1:]:
        if p[1] > out[-1][1]:
            out.append(p)
    return out


def interp(front, x, mode):
    if x >= front[-1][0]:
        return front[-1][1]
    if mode == "step":
        return max(y for xx, y in front if xx <= x + 1e-12)
    for (x0, y0), (x1, y1) in zip(front, front[1:]):
        if x0 <= x <= x1:
            return y0 + (y1 - y0) * (x - x0) / (x1 - x0) if x1 > x0 else max(y0, y1)
    return 0.0


def fam_pts(fam, reading, form):
    return [(v[reading]["fuel"], v[reading][form]) for (f, l), v in S.items() if f == fam]


print("=== reproduce the headline (staircase + linear interpolation, gap at sighted's own points)")
for reading in ("noknock", "knock"):
    for form in ("mean", "ev"):
        fb = stair(fam_pts("blind", reading, form))
        out = []
        for r in RATIOS:
            fs = stair(fam_pts(f"s{r}", reading, form))
            gaps = [(x, y - interp(fb, x, "lin")) for x, y in fs if x > 0]
            g = max(gaps, key=lambda t: t[1])
            out.append(f"r={r}: {g[1]:+.3f} at {g[0]:.3f}%")
        print(f"  {reading:8s} {form:5s} " + "   ".join(out))

print("\n=== mixing assumption: blind frontier as step (no mixing), linear-between-staircase (as hw_opt), concave hull (random mixing)")
print("    and both sides hulled (sighted hull minus blind hull, max over x in (0, sighted's last x])")
for reading in ("noknock", "knock"):
    for form in ("mean", "ev"):
        bp = fam_pts("blind", reading, form)
        fb_s, fb_h = stair(bp), hull(bp)
        for r in RATIOS:
            sp = fam_pts(f"s{r}", reading, form)
            fs = stair(sp)
            fh = hull(sp)
            res = {}
            for mode, fb, m in (("step", fb_s, "step"), ("lin", fb_s, "lin"), ("hull", fb_h, "lin")):
                gaps = [(x, y - interp(fb, x, m)) for x, y in fs if x > 0]
                res[mode] = max(gaps, key=lambda t: t[1])
            xs = np.linspace(1e-3, fh[-1][0], 4000)
            hh = max(((x, interp(fh, x, "lin") - interp(fb_h, x, "lin")) for x in xs), key=lambda t: t[1])
            print(f"  {reading:8s} {form:5s} r={r}: step {res['step'][1]:+.3f}@{res['step'][0]:.2f}  "
                  f"lin {res['lin'][1]:+.3f}@{res['lin'][0]:.2f}  blind-hull {res['hull'][1]:+.3f}@{res['hull'][0]:.2f}  "
                  f"hull-vs-hull {hh[1]:+.3f}@{hh[0]:.2f}")
        print(f"     blind hull vertices ({reading},{form}): {[(round(a, 3), round(b, 3)) for a, b in fb_h]}")

    # evaluate form of the reward-chosen sets
